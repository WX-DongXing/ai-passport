cmake_minimum_required(VERSION 3.20)
project(offduty_render C)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${REPO}/managed_components/lvgl__lvgl" lvgl)
file(GLOB FONTS "${REPO}/assets/fonts/offduty/*.c")
add_executable(offduty_render render.c "${REPO}/main/offduty_ui.c" "${REPO}/main/offduty_model.c" ${FONTS} "${REPO}/assets/images/offduty/od_stamps.c")
target_sources(offduty_render PRIVATE "${REPO}/assets/images/offduty/od_pets.c" "${REPO}/assets/images/offduty/od_boot_pet.c")
target_include_directories(offduty_render PRIVATE "${REPO}/main")
target_link_libraries(offduty_render PRIVATE lvgl)

target_compile_definitions(offduty_render PRIVATE OD_UI_VALIDATE=1)

option(OD_STAMP_REFERENCE "Render stamps procedurally for asset regeneration" OFF)
if(OD_STAMP_REFERENCE)
    target_compile_definitions(offduty_render PRIVATE OD_STAMP_REFERENCE=1)
endif()
option(OD_PET_REFERENCE "Generate transparent original pet assets" OFF)
if(OD_PET_REFERENCE)
    add_executable(offduty_pet_assets pet_assets.c "${REPO}/main/offduty_ui.c" "${REPO}/main/offduty_model.c" ${FONTS} "${REPO}/assets/images/offduty/od_stamps.c")
    target_sources(offduty_pet_assets PRIVATE "${REPO}/assets/images/offduty/od_pets.c" "${REPO}/assets/images/offduty/od_boot_pet.c")
    target_include_directories(offduty_pet_assets PRIVATE "${REPO}/main")
    target_compile_definitions(offduty_pet_assets PRIVATE OD_PET_REFERENCE=1)
    target_link_libraries(offduty_pet_assets PRIVATE lvgl)
endif()
